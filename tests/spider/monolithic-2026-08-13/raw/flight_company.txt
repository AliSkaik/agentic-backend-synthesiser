### PostgreSQL Schema

```sql
CREATE TABLE airports (
    id SERIAL PRIMARY KEY,
    country VARCHAR(255) NOT NULL,
    city VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL
);

CREATE TABLE vehicles (
    flight_number VARCHAR(10) PRIMARY KEY,
    velocity INT NOT NULL,
    altitude INT NOT NULL,
    pilot_id INT,
    airport_id INT,
    FOREIGN KEY (pilot_id) REFERENCES pilots(id),
    FOREIGN KEY (airport_id) REFERENCES airports(id)
);

CREATE TABLE pilots (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL
);
```

### Express Router

```javascript
import express from 'express';
import pool from '../db.js';

const router = express.Router();

// 1. How many flights have a velocity larger than 200?
router.get('/flights/velocity', async (req, res) => {
    try {
        const result = await pool.query('SELECT COUNT(*) FROM vehicles WHERE velocity > 200');
        res.json({ count: result.rows[0].count });
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 2. List the vehicle flight number, date and pilot of all the flights, ordered by altitude.
router.get('/flights', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT v.flight_number, v.date, p.name AS pilot
            FROM vehicles v
            JOIN pilots p ON v.pilot_id = p.id
            ORDER BY v.altitude;
        `);
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 3. List the id, country, city and name of the airports ordered alphabetically by the name.
router.get('/airports', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT id, country, city, name
            FROM airports
            ORDER BY name;
        `);
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 4. What is the maximum group equity shareholding of the companies?
router.get('/companies/max-shareholding', async (req, res) => {
    try {
        const result = await pool.query('SELECT MAX(group_equity_shareholding) FROM companies');
        res.json({ maxShareholding: result.rows[0].max });
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 5. What is the velocity of the pilot named 'Thompson'?
router.get('/pilots/thompson/velocity', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT v.velocity
            FROM vehicles v
            JOIN pilots p ON v.pilot_id = p.id
            WHERE p.name = 'Thompson';
        `);
        if (result.rows.length > 0) {
            res.json({ velocity: result.rows[0].velocity });
        } else {
            res.status(404).json({ error: 'Pilot not found' });
        }
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

export default router;
```

### Explanation

1. **Database Schema**:
   - `airports`: Stores information about airports.
   - `vehicles`: Stores information about vehicles, including their velocity and altitude.
   - `pilots`: Stores information about pilots.

2. **Express Router**:
   - `/flights/velocity`: Returns the count of flights with a velocity larger than 200.
   - `/flights`: Lists all flights ordered by altitude.
   - `/airports`: Lists all airports ordered alphabetically by name.
   - `/companies/max-shareholding`: Returns the maximum group equity shareholding of companies (assuming there is a `companies` table).
   - `/pilots/thompson/velocity`: Returns the velocity of the pilot named 'Thompson'.
