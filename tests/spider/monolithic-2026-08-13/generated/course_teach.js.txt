import express from 'express';
import { pool } from '../db.js';

const router = express.Router();

// Endpoint to get the total count of teachers
router.get('/count', async (req, res) => {
    try {
        const result = await pool.query('SELECT COUNT(*) FROM teachers');
        res.json({ count: result.rows[0].count });
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// Endpoint to list the names of teachers in ascending order of age
router.get('/names-ascending', async (req, res) => {
    try {
        const result = await pool.query('SELECT name FROM teachers ORDER BY age ASC');
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// Endpoint to get the age and hometown of teachers
router.get('/age-hometown', async (req, res) => {
    try {
        const result = await pool.query('SELECT age, hometown FROM teachers');
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

export default router;
