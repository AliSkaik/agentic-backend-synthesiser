import express from 'express';
import pool from '../db.js';

const router = express.Router();

// 1. List the names of all the customers in alphabetical order.
router.get('/customers', async (req, res) => {
    try {
        const result = await pool.query('SELECT first_name, last_name FROM customers ORDER BY first_name, last_name');
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 2. Find all the policy type codes associated with the customer "Dayana Robel".
router.get('/customers/dayana-robel/policy-types', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT DISTINCT policy_type_code
            FROM policies
            JOIN customers ON policies.customer_id = customers.customer_id
            WHERE first_name = 'Dayana' AND last_name = 'Robel'
        `);
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 3. Which type of policy is most frequently used? Give me the policy type code.
router.get('/policy-types/most-frequent', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT policy_type_code
            FROM policies
            GROUP BY policy_type_code
            ORDER BY COUNT(*) DESC
            LIMIT 1
        `);
        res.json(result.rows[0]);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

export default router;
