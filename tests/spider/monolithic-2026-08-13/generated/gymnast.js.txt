import express from 'express';
import pool from '../db.js';

const router = express.Router();

// Endpoint to get the total number of gymnasts
router.get('/count', async (req, res) => {
    try {
        const result = await pool.query('SELECT COUNT(*) FROM gymnasts');
        res.json({ count: result.rows[0].count });
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// Endpoint to list the total points of gymnasts in descending order
router.get('/total-points', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT 
                id,
                name,
                floor_exercise_points + vault_points + bars_points + beam_points AS total_points
            FROM 
                gymnasts
            ORDER BY 
                total_points DESC;
        `);
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// Endpoint to list the total points of gymnasts in descending order of floor exercise points
router.get('/total-points-floor', async (req, res) => {
    try {
        const result = await pool.query(`
            SELECT 
                id,
                name,
                floor_exercise_points + vault_points + bars_points + beam_points AS total_points
            FROM 
                gymnasts
            ORDER BY 
                floor_exercise_points DESC;
        `);
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

export default router;
