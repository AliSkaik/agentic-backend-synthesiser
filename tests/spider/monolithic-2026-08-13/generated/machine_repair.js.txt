import express from 'express';
import pool from '../db.js';

const router = express.Router();

// 1. How many technicians are there?
router.get('/count', async (req, res) => {
    try {
        const result = await pool.query('SELECT COUNT(*) FROM technicians');
        res.json({ count: result.rows[0].count });
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 2. List the names of technicians in ascending order of age
router.get('/names-by-age', async (req, res) => {
    try {
        const result = await pool.query('SELECT name FROM technicians ORDER BY age ASC');
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

// 3. What are the team and starting year of technicians?
router.get('/team-and-starting-year', async (req, res) => {
    try {
        const result = await pool.query('SELECT name, team, starting_year FROM technicians');
        res.json(result.rows);
    } catch (error) {
        res.status(500).json({ error: 'Internal Server Error' });
    }
});

export default router;
